def _bytes_to_pil(b):
    return Image.open(io.BytesIO(b)).convert("RGB")

def _path_to_pil(p):
    return Image.open(p).convert("RGB")

def _dict_to_pil(d):
    if isinstance(d, dict):
        if "bytes" in d and isinstance(d["bytes"], (bytes, bytearray)):
            return _bytes_to_pil(d["bytes"])
        if "path" in d and isinstance(d["path"], str):
            return _path_to_pil(d["path"])
    raise ValueError("Unsupported image dict")

def guess_pil_from_row(row_dict):
    """
    Fallback：不用 crop 的情况下，从原始 image.bytes / image.path 读取。
    """
    if "image.bytes" in row_dict and isinstance(row_dict["image.bytes"], (bytes, bytearray)):
        return _bytes_to_pil(row_dict["image.bytes"])

    if "image.path" in row_dict and isinstance(row_dict["image.path"], str):
        p = row_dict["image.path"]
        if len(p) > 0 and os.path.exists(p):
            return _path_to_pil(p)

    if "image" in row_dict and isinstance(row_dict["image"], dict):
        return _dict_to_pil(row_dict["image"])

    # 最后再试一次 image.path，即使文件不存在也试一下
    if "image.path" in row_dict and isinstance(row_dict["image.path"], str):
        try:
            return _path_to_pil(row_dict["image.path"])
        except Exception:
            pass

    raise FileNotFoundError("No usable image source (bytes/path/dict).")

def safe_row_to_pil(row):
    """
    优先使用裁剪结果（crop_path / crop_bytes），否则退回原图。
    row: df_train.iloc[i] 这种 Series
    """
    # 1) on-disk crop
    cp = row.get("crop_path", None)
    if isinstance(cp, str) and len(cp) and os.path.exists(cp):
        try:
            return _path_to_pil(cp)
        except Exception:
            pass

    # 2) in-memory crop bytes
    cb = row.get("crop_bytes", None)
    if isinstance(cb, (bytes, bytearray)):
        try:
            return _bytes_to_pil(cb)
        except Exception:
            pass

    # 3) fallback to original
    return guess_pil_from_row(row.to_dict())

class CarsParquet(Dataset):
    def __init__(self, df: pd.DataFrame, transform):
        self.df = df.reset_index(drop=True)
        self.transform = transform

    def __len__(self):
        return len(self.df)

    def __getitem__(self, i):
        row = self.df.iloc[i]
        y = int(row["label"])

        # 优先用裁剪结果
        img = safe_row_to_pil(row)
        x = self.transform(img)
        return x, y

def make_model(name: str, ncls: int, freeze_backbone: bool = True):
    name = name.lower()

    if name == "resnet50":
        m = tvm.resnet50(weights=tvm.ResNet50_Weights.IMAGENET1K_V2)
        if freeze_backbone:
            for p in m.parameters():
                p.requires_grad = False
        in_ch = m.fc.in_features
        m.fc = nn.Linear(in_ch, ncls)
        for p in m.fc.parameters():
            p.requires_grad = True
        return m

    if name == "efficientnet_b0":
        m = tvm.efficientnet_b0(weights=tvm.EfficientNet_B0_Weights.IMAGENET1K_V1)
        if freeze_backbone:
            for p in m.parameters():
                p.requires_grad = False
        in_ch = m.classifier[1].in_features
        m.classifier[1] = nn.Linear(in_ch, ncls)
        for p in m.classifier.parameters():
            p.requires_grad = True
        return m

    if name == "convnext_tiny":
        m = tvm.convnext_tiny(weights=tvm.ConvNeXt_Tiny_Weights.IMAGENET1K_V1)
        if freeze_backbone:
            for p in m.parameters():
                p.requires_grad = False
        in_ch = m.classifier[2].in_features
        m.classifier[2] = nn.Linear(in_ch, ncls)
        for p in m.classifier.parameters():
            p.requires_grad = True
        return m

    raise ValueError(f"Unknown backbone: {name}")

def acc_top1(logits, y):
    return (logits.argmax(1) == y).float().mean().item()

def train_one_epoch(model, loader):
    model.train()
    loss_sum, acc_sum = 0.0, 0.0

    for x, y in loader:
        x = x.to(device)
        y = y.to(device)

        optimizer.zero_grad(set_to_none=True)

        with torch.cuda.amp.autocast(enabled=USE_AMP):
            logits = model(x)
            loss = criterion(logits, y)

        scaler.scale(loss).backward()

        if GRAD_CLIP_NORM and GRAD_CLIP_NORM > 0:
            scaler.unscale_(optimizer)
            nn.utils.clip_grad_norm_(model.parameters(), GRAD_CLIP_NORM)

        scaler.step(optimizer)
        scaler.update()

        loss_sum += loss.item()
        acc_sum  += acc_top1(logits.detach(), y)

    return loss_sum / len(loader), acc_sum / len(loader)

@torch.no_grad()
def evaluate(model, loader):
    model.eval()
    loss_sum, acc_sum = 0.0, 0.0
    preds, labels = [], []

    for x, y in loader:
        x = x.to(device)
        y = y.to(device)

        with torch.cuda.amp.autocast(enabled=USE_AMP):
            logits = model(x)
            loss = criterion(logits, y)

        loss_sum += loss.item()
        acc_sum  += acc_top1(logits, y)

        preds.append(logits.argmax(1).cpu())
        labels.append(y.cpu())

    preds  = torch.cat(preds).numpy()
    labels = torch.cat(labels).numpy()
    return loss_sum / len(loader), acc_sum / len(loader), preds, labels

def confusion_matrix_numpy(preds, labels, ncls):
    cm = np.zeros((ncls, ncls), dtype=np.int64)
    for p, t in zip(preds, labels):
        cm[t, p] += 1
    return cm

def plot_cm(cm, path):
    fig, ax = plt.subplots(figsize=(7, 7))
    im = ax.imshow(cm, cmap="Blues")
    ax.figure.colorbar(im, ax=ax)
    ax.set_xlabel("Pred")
    ax.set_ylabel("True")
    ax.set_title("Confusion Matrix")
    plt.tight_layout()
    plt.savefig(path)
    plt.close(fig)

def load_best_model(model, device):
    # 你现在用的路径保持不变
    standard_path = "31/best_stanford_cars_transfer_model.pth"
    exp_path = OUT_DIR / "best_stanford_cars_transfer_model.pth"

    if os.path.exists(standard_path):
        print(f"Loading model from {standard_path}")
        model.load_state_dict(torch.load(standard_path, map_location=device))
        return True
    elif exp_path.exists():
        print(f"Loading model from {exp_path}")
        model.load_state_dict(torch.load(exp_path, map_location=device))
        return True
    else:
        print("No checkpoint found, using current model state.")
        return False

def current_lr(optim):
    return optim.param_groups[0]["lr"]

try:
    for ep in range(1, EPOCHS + 1):
        t0 = time.time()

        tr_loss, tr_acc = train_one_epoch(model, train_loader)
        va_loss, va_acc, va_preds, va_labels = evaluate(model, test_loader)

        scheduler.step()
        dt = time.time() - t0
        lr_now = current_lr(optimizer)

        print(
            f"Epoch {ep:02d}/{EPOCHS} | "
            f"lr {lr_now:.6f} | "
            f"train {tr_loss:.4f}/{tr_acc:.4f} | "
            f"val {va_loss:.4f}/{va_acc:.4f} | "
            f"time {dt:.1f}s"
        )

        history.append({
            "epoch": ep,
            "lr": lr_now,
            "train_loss": tr_loss,
            "train_acc": tr_acc,
            "val_loss": va_loss,
            "val_acc": va_acc,
            "secs": dt,
        })

        improved = va_acc > best_acc or np.isnan(best_acc)
        if improved:
            best_acc = va_acc
            no_imp = 0
            torch.save(
                model.state_dict(),
                "31/best_stanford_cars_transfer_model.pth"
            )
        else:
            no_imp += 1
            if no_imp >= EARLY_STOP_PATIENCE:
                print("Early stop triggered.")
                break

        with open(OUT_DIR / "history.json", "w") as f:
            json.dump(history, f, indent=2)

except KeyboardInterrupt:
